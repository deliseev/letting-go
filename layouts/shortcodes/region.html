{{- /* Renders one named region from a file in examples/, delimited by // region: ... // endregion: See CLAUDE.md at the repo root for the region-marker convention. Fails the build (does not render an empty block) if the file is missing or the named region isn't in it, so a stale reference can't silently ship. @param {string} file Path of the source file, relative to examples/. @param {string} name Region name to extract. @param {string} [lang] Highlight language. Defaults to "go". @example {{< region file="foundations/value-receiver.go" name="value-receiver" >}} */}} {{- $file := .Get "file" -}} {{- $name := .Get "name" -}} {{- $lang := .Get "lang" | default "go" -}} {{- if not $file -}} {{- errorf "%q requires a %q parameter. %s" .Name "file" .Position -}} {{- else if not $name -}} {{- errorf "%q requires a %q parameter. %s" .Name "name" .Position -}} {{- else -}} {{- $path := printf "examples/%s" $file -}} {{- if not (os.FileExists $path) -}} {{- errorf "%q: no such file %q. %s" .Name $path .Position -}} {{- else -}} {{- $startMarker := printf "// region: %s" $name -}} {{- $endMarker := printf "// endregion: %s" $name -}} {{- $collecting := false -}} {{- $found := false -}} {{- $lines := slice -}} {{- range strings.Split (os.ReadFile $path) "\n" -}} {{- $trimmed := strings.TrimSpace . -}} {{- if eq $trimmed $startMarker -}} {{- if $found -}} {{- errorf "%q: region %q appears more than once in %q. %s" $.Name $name $path $.Position -}} {{- end -}} {{- $collecting = true -}} {{- $found = true -}} {{- else if eq $trimmed $endMarker -}} {{- $collecting = false -}} {{- else if $collecting -}} {{- $lines = $lines | append . -}} {{- end -}} {{- end -}} {{- if not $found -}} {{- errorf "%q: region %q not found in %q. %s" .Name $name $path .Position -}} {{- else if $collecting -}} {{- errorf "%q: region %q in %q has no matching %q. %s" .Name $name $path $endMarker .Position -}} {{- else -}} {{- /* strings.TrimSpace strips leading/trailing blank lines too, since a blank line is just whitespace either side of \n */ -}} {{- $content := printf "%s\n" (strings.TrimSpace (delimit $lines "\n")) -}}
{{- partial "components/codeblock" (dict "filename" $file "lang" $lang "content" $content "options" (dict)) -}} {{- if or (eq site.Params.highlight.copy.enable nil) (site.Params.highlight.copy.enable) -}} {{- partialCached "components/codeblock-copy-button" (dict "filename" $file) $path -}} {{- end -}}
{{- end -}} {{- end -}} {{- end -}}
